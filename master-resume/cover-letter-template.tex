% Glyph-to-unicode mapping improves copy-paste/ATS text extraction, but the
% primitives it relies on only exist under pdfTeX (not XeTeX/LuaTeX engines
% some compilers, e.g. tectonic, default to).
%
% On those engines the mapping is simply absent, and xdvipdfmx emits no usable
% ToUnicode entry for the f-ligatures: "efficiency" is drawn with a single
% U+FB00 glyph and pdftotext extracts it as "e<ffi>ciency", so an ATS keyword
% match on the word silently fails. Suppressing the common f-ligatures is the
% equivalent fix there - nothing else in the rendering changes.
\ifpdftex
  \input{glyphtounicode}
  \pdfgentounicode=1
\else
  \usepackage{fontspec}
  \setmainfont{lmroman10-regular.otf}[
    BoldFont       = lmroman10-bold.otf,
    ItalicFont     = lmroman10-italic.otf,
    BoldItalicFont = lmroman10-bolditalic.otf,
    SmallCapsFont  = lmromancaps10-regular.otf,
    Ligatures      = NoCommon]
\fi
